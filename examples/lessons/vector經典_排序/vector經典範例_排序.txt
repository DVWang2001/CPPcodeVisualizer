#include <bits/stdc++.h>
using namespace std;
void SORT(vector<int>&arr) {
    for (int i = 0; i < arr.size(); i++) {
        for (int j = 0; j < arr.size()-i-1; j++) {
            if (arr[j] > arr[j+1]) {
                swap(arr[j],arr[j+1]);
            }
        }
    }
}
int main() {
    vector<int>arr;
    for (int i = 0; i < 10; i++) {
        arr.push_back(10-i-1);
    }
    SORT(arr);
}
